-- ============================================================
-- 1. AUDIT: what is stale and open right now (read-only)
-- ============================================================
SELECT d.id, d.status, d.giveaway_id, g.name, d.entry_closes_at,
       (SELECT COUNT(*) FROM tickets t WHERE t.draw_id = d.id) AS tickets,
       (SELECT COUNT(*) FROM draw_winners w WHERE w.draw_id = d.id) AS winners
FROM draws d LEFT JOIN giveaways g ON g.id = d.giveaway_id
WHERE d.kind = 'custom' AND d.status = 'entry_open'
ORDER BY d.created_at;

-- ============================================================
-- 2. BACKFILL closing times from each giveaway's schedule,
--    so old draws without one close on time by themselves
-- ============================================================
UPDATE draws d SET entry_closes_at = g.scheduled_at
FROM giveaways g
WHERE d.giveaway_id = g.id
  AND d.status = 'entry_open'
  AND d.entry_closes_at IS NULL
  AND g.scheduled_at IS NOT NULL;

-- ============================================================
-- 3. CLOSE stale draws that have NO tickets (nothing to draw,
--    no entries lost). Draws WITH tickets are left for the
--    worker, which draws their winners fairly on next tick.
-- ============================================================
UPDATE draws d SET status = 'closed', drawn_at = NOW()
FROM giveaways g
WHERE d.giveaway_id = g.id
  AND d.kind = 'custom'
  AND d.status = 'entry_open'
  AND d.id NOT IN (SELECT draw_id FROM tickets)
  AND (d.entry_closes_at <= NOW()
    OR (d.entry_closes_at IS NULL AND (
      (g.interval_minutes IS NOT NULL AND g.ends_at IS NOT NULL AND g.ends_at <= NOW())
      OR (g.interval_minutes IS NULL AND g.scheduled_at IS NOT NULL AND g.scheduled_at <= NOW())
    )));
